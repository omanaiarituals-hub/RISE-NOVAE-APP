'use client'

import { useState } from 'react'
import Link from 'next/link'
import { supabase } from '@/lib/supabase/client'
import type { NovaPlanResult, NovaProviderPreference } from '@/lib/nova-ai/types'

const EXAMPLES = [
  'Je dois envoyer mon dossier à la CPAM avant vendredi.',
  'Mardi à 14 h, j’ai rendez-vous chez le dentiste avec Inaya.',
  'J’ai reçu une facture d’électricité de 126 euros à payer avant le 10 août.',
  'Note que cette semaine j’ai les enfants mercredi, jeudi et vendredi.',
]

export default function NovaLabClient({ userEmail }: { userEmail: string }) {
  const [message, setMessage] = useState(EXAMPLES[0])
  const [provider, setProvider] = useState<NovaProviderPreference>('auto')
  const [result, setResult] = useState<NovaPlanResult | null>(null)
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  async function analyse() {
    if (!message.trim() || loading) return
    setLoading(true)
    setError('')
    setResult(null)

    try {
      const { data } = await supabase.auth.getSession()
      const token = data.session?.access_token
      if (!token) throw new Error('Session introuvable. Reconnecte-toi.')

      const response = await fetch('/api/nova/plan', {
        method: 'POST',
        headers: {
          'content-type': 'application/json',
          authorization: `Bearer ${token}`,
        },
        body: JSON.stringify({ message, provider }),
      })

      const payload = await response.json()
      if (!response.ok) throw new Error(payload.message || payload.error || 'Analyse impossible.')
      setResult(payload as NovaPlanResult)
    } catch (caught) {
      setError(caught instanceof Error ? caught.message : 'Une erreur est survenue.')
    } finally {
      setLoading(false)
    }
  }

  return (
    <main className="min-h-screen bg-[#F7F5F1] px-4 py-8 text-[#282522]">
      <div className="mx-auto max-w-5xl">
        <div className="mb-8 flex flex-wrap items-start justify-between gap-4">
          <div>
            <p className="mb-2 text-xs font-semibold uppercase tracking-[0.22em] text-[#796F68]">Laboratoire interne</p>
            <h1 className="font-serif text-3xl font-semibold">Nova V2 · Plan d’action</h1>
            <p className="mt-2 max-w-2xl text-sm leading-6 text-[#625B55]">
              Cette page analyse une demande et prépare des propositions. Elle ne crée aucune tâche, aucun rappel et aucune donnée.
            </p>
          </div>
          <Link href="/" className="rounded-full border border-[#CFC7BF] bg-white px-4 py-2 text-sm">Retour</Link>
        </div>

        <section className="rounded-2xl border border-[#D7D0C8] bg-white p-5 shadow-sm">
          <div className="mb-4 flex flex-wrap items-center justify-between gap-3 text-xs text-[#766D66]">
            <span>Compte : {userEmail}</span>
            <span className="rounded-full bg-[#EFEAE5] px-3 py-1 font-medium">Mode lecture seule</span>
          </div>

          <label className="mb-2 block text-sm font-semibold" htmlFor="nova-message">Situation à analyser</label>
          <textarea
            id="nova-message"
            value={message}
            onChange={(event) => setMessage(event.target.value)}
            rows={6}
            className="w-full rounded-xl border border-[#CBC3BB] bg-[#FCFBF9] p-4 text-base leading-6 outline-none focus:border-[#7B6F66]"
          />

          <div className="mt-3 flex flex-wrap gap-2">
            {EXAMPLES.map((example) => (
              <button
                type="button"
                key={example}
                onClick={() => setMessage(example)}
                className="rounded-full border border-[#DDD6CF] px-3 py-1.5 text-xs hover:bg-[#F4F0EC]"
              >
                {example.slice(0, 45)}{example.length > 45 ? '…' : ''}
              </button>
            ))}
          </div>

          <div className="mt-5 flex flex-wrap items-end gap-3">
            <label className="text-sm">
              <span className="mb-1 block font-semibold">Fournisseur</span>
              <select
                value={provider}
                onChange={(event) => setProvider(event.target.value as NovaProviderPreference)}
                className="rounded-lg border border-[#CBC3BB] bg-white px-3 py-2"
              >
                <option value="auto">Automatique avec secours</option>
                <option value="anthropic">Anthropic uniquement</option>
                <option value="openai">OpenAI uniquement</option>
              </select>
            </label>
            <button
              type="button"
              onClick={analyse}
              disabled={loading || !message.trim()}
              className="rounded-lg bg-[#332E2A] px-5 py-2.5 font-semibold text-white disabled:cursor-not-allowed disabled:opacity-50"
            >
              {loading ? 'Analyse en cours…' : 'Analyser sans exécuter'}
            </button>
          </div>

          {error && <p className="mt-4 rounded-xl border border-[#C8A6A1] bg-[#F8EFEE] p-3 text-sm text-[#7B2921]">{error}</p>}
        </section>

        {result && (
          <div className="mt-6 space-y-5">
            <section className="rounded-2xl border border-[#D7D0C8] bg-white p-5 shadow-sm">
              <div className="flex flex-wrap justify-between gap-3">
                <div>
                  <p className="text-xs uppercase tracking-wider text-[#796F68]">Réponse proposée par Nova</p>
                  <p className="mt-2 text-lg leading-7">{result.plan.assistant_message}</p>
                </div>
                <div className="text-right text-xs leading-5 text-[#716860]">
                  <p>{result.provider} · {result.model}</p>
                  <p>{result.durationMs} ms</p>
                  <p>Essais : {result.attemptedProviders.join(' → ')}</p>
                </div>
              </div>
            </section>

            <div className="grid gap-5 md:grid-cols-2">
              <section className="rounded-2xl border border-[#D7D0C8] bg-white p-5 shadow-sm">
                <h2 className="font-serif text-xl font-semibold">Ce que Nova a compris</h2>
                <p className="mt-3 text-sm leading-6"><strong>Intention :</strong> {result.plan.intent}</p>
                <p className="text-sm leading-6"><strong>Confiance :</strong> {Math.round(result.plan.confidence * 100)} %</p>
                <p className="mt-3 text-sm leading-6">{result.plan.summary}</p>
                <pre className="mt-4 max-h-80 overflow-auto rounded-xl bg-[#F4F1ED] p-3 text-xs leading-5">{JSON.stringify(result.plan.extracted_data, null, 2)}</pre>
              </section>

              <section className="rounded-2xl border border-[#D7D0C8] bg-white p-5 shadow-sm">
                <h2 className="font-serif text-xl font-semibold">Actions proposées</h2>
                <div className="mt-3 space-y-3">
                  {result.plan.proposed_actions.length === 0 && <p className="text-sm text-[#716860]">Aucune action proposée.</p>}
                  {result.plan.proposed_actions.map((action) => (
                    <article key={action.id} className="rounded-xl border border-[#E1DBD5] p-3">
                      <div className="flex justify-between gap-3">
                        <strong className="text-sm">{action.title}</strong>
                        <span className="text-xs uppercase text-[#766D66]">{action.engine}</span>
                      </div>
                      <p className="mt-1 text-sm leading-5 text-[#5E5751]">{action.reason}</p>
                      <p className="mt-2 text-xs">Validation : {action.requires_confirmation ? 'obligatoire' : 'non requise'} · Risque : {action.risk}</p>
                    </article>
                  ))}
                </div>
              </section>
            </div>

            <section className="rounded-2xl border border-[#D7D0C8] bg-white p-5 shadow-sm">
              <h2 className="font-serif text-xl font-semibold">Résultat technique complet</h2>
              <pre className="mt-3 max-h-[34rem] overflow-auto rounded-xl bg-[#292623] p-4 text-xs leading-5 text-[#F7F3EF]">{JSON.stringify(result, null, 2)}</pre>
            </section>
          </div>
        )}
      </div>
    </main>
  )
}
