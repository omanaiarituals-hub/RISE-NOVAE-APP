'use client'

import { useState } from 'react'
import Link from 'next/link'
import { supabase } from '@/lib/supabase/client'
import type { NovaPlanResult } from '@/lib/nova-ai/types'

const EXAMPLES = [
  'Je dois envoyer mon dossier à la CPAM avant vendredi.',
  'Mardi à 14 h, j’ai rendez-vous chez le dentiste avec Inaya.',
  'J’ai reçu une facture d’électricité de 126 euros à payer avant le 10 août.',
  'Note que cette semaine j’ai les enfants mercredi, jeudi et vendredi.',
]

export default function NovaLabClient() {
  const [message, setMessage] = useState(EXAMPLES[0])
  const [result, setResult] = useState<NovaPlanResult | null>(null)
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  async function analyse() {
    if (!message.trim() || loading) return

    setLoading(true)
    setError('')
    setResult(null)

    try {
      const { data } = await supabase.auth.getSession()
      const token = data.session?.access_token

      if (!token) {
        throw new Error('Ta session a expiré. Reconnecte-toi à NOVAÉ.')
      }

      const response = await fetch('/api/nova/plan', {
        method: 'POST',
        headers: {
          'content-type': 'application/json',
          authorization: `Bearer ${token}`,
        },
        body: JSON.stringify({
          message,
          provider: 'auto',
        }),
      })

      const payload = await response.json()

      if (!response.ok) {
        throw new Error(
          'Nova n’a pas pu analyser cette demande pour le moment. Réessaie dans quelques instants.'
        )
      }

      setResult(payload as NovaPlanResult)
    } catch (caught) {
      setError(
        caught instanceof Error
          ? caught.message
          : 'Une erreur est survenue pendant l’analyse.'
      )
    } finally {
      setLoading(false)
    }
  }

  return (
    <main className="min-h-screen bg-[#F7F5F1] px-4 py-8 text-[#282522]">
      <div className="mx-auto max-w-4xl">
        <header className="mb-8 flex flex-wrap items-start justify-between gap-4">
          <div>
            <p className="mb-2 text-xs font-semibold uppercase tracking-[0.22em] text-[#796F68]">
              Laboratoire privé
            </p>

            <h1 className="font-serif text-3xl font-semibold">
              Nova V2
            </h1>

            <p className="mt-2 max-w-2xl text-sm leading-6 text-[#625B55]">
              Nova analyse la situation et prépare la suite. Aucune donnée,
              tâche ou notification n’est enregistrée pendant ce test.
            </p>
          </div>

          <Link
            href="/"
            className="rounded-full border border-[#CFC7BF] bg-white px-4 py-2 text-sm"
          >
            Retour
          </Link>
        </header>

        <section className="rounded-2xl border border-[#D7D0C8] bg-white p-5 shadow-sm">
          <div className="mb-4 flex justify-end">
            <span className="rounded-full bg-[#EFEAE5] px-3 py-1 text-xs font-medium text-[#625B55]">
              Mode test, sans exécution
            </span>
          </div>

          <label
            className="mb-2 block text-sm font-semibold"
            htmlFor="nova-message"
          >
            Que veux-tu confier à Nova ?
          </label>

          <textarea
            id="nova-message"
            value={message}
            onChange={(event) => setMessage(event.target.value)}
            rows={6}
            className="w-full rounded-xl border border-[#CBC3BB] bg-[#FCFBF9] p-4 text-base leading-6 outline-none focus:border-[#7B6F66]"
          />

          <div className="mt-3 flex flex-wrap gap-2">
            {EXAMPLES.map((example) => (
              <button
                type="button"
                key={example}
                onClick={() => setMessage(example)}
                className="rounded-full border border-[#DDD6CF] px-3 py-1.5 text-xs hover:bg-[#F4F0EC]"
              >
                {example.slice(0, 45)}
                {example.length > 45 ? '…' : ''}
              </button>
            ))}
          </div>

          <button
            type="button"
            onClick={analyse}
            disabled={loading || !message.trim()}
            className="mt-5 rounded-lg bg-[#332E2A] px-5 py-2.5 font-semibold text-white disabled:cursor-not-allowed disabled:opacity-50"
          >
            {loading ? 'Nova analyse…' : 'Demander à Nova'}
          </button>

          {error && (
            <p className="mt-4 rounded-xl border border-[#C8A6A1] bg-[#F8EFEE] p-3 text-sm text-[#7B2921]">
              {error}
            </p>
          )}
        </section>

        {result && (
          <div className="mt-6 space-y-5">
            <section className="rounded-2xl border border-[#D7D0C8] bg-white p-6 shadow-sm">
              <p className="text-xs font-semibold uppercase tracking-[0.16em] text-[#796F68]">
                Réponse de Nova
              </p>

              <p className="mt-3 font-serif text-xl leading-8">
                {result.plan.assistant_message}
              </p>
            </section>

            <section className="rounded-2xl border border-[#D7D0C8] bg-white p-6 shadow-sm">
              <h2 className="font-serif text-xl font-semibold">
                Ce que Nova a compris
              </h2>

              <p className="mt-3 text-sm leading-6 text-[#514B46]">
                {result.plan.summary}
              </p>
            </section>

            {result.plan.missing_information.length > 0 && (
              <section className="rounded-2xl border border-[#D7D0C8] bg-white p-6 shadow-sm">
                <h2 className="font-serif text-xl font-semibold">
                  Informations à préciser
                </h2>

                <div className="mt-4 space-y-3">
                  {result.plan.missing_information.map((information, index) => (
                    <div
                      key={`${information.field}-${index}`}
                      className="rounded-xl bg-[#F4F1ED] p-4 text-sm leading-6"
                    >
                      {information.question}
                    </div>
                  ))}
                </div>
              </section>
            )}

            <section className="rounded-2xl border border-[#D7D0C8] bg-white p-6 shadow-sm">
              <h2 className="font-serif text-xl font-semibold">
                Ce que Nova peut préparer
              </h2>

              <div className="mt-4 space-y-3">
                {result.plan.proposed_actions.length === 0 && (
                  <p className="text-sm text-[#716860]">
                    Aucune action particulière n’est nécessaire.
                  </p>
                )}

                {result.plan.proposed_actions.map((action) => (
                  <article
                    key={action.id}
                    className="rounded-xl border border-[#E1DBD5] p-4"
                  >
                    <div className="flex flex-wrap items-start justify-between gap-3">
                      <strong className="text-sm">
                        {action.title}
                      </strong>

                      {action.requires_confirmation && (
                        <span className="rounded-full bg-[#EFEAE5] px-3 py-1 text-xs text-[#625B55]">
                          À valider
                        </span>
                      )}
                    </div>

                    <p className="mt-2 text-sm leading-6 text-[#5E5751]">
                      {action.reason}
                    </p>
                  </article>
                ))}
              </div>
            </section>

            <p className="text-center text-xs leading-5 text-[#766D66]">
              Rien ne sera exécuté tant que la validation n’aura pas été ajoutée
              au prochain chantier.
            </p>
          </div>
        )}
      </div>
    </main>
  )
}
